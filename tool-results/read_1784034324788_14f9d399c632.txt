   813→// Helper to get a nested value from an object using a dot-path
   814→function getNestedValue(obj: unknown, path: string): string | undefined {
   815→  const keys = path.split(".")
   816→  let current: unknown = obj
   817→  for (const key of keys) {
   818→    if (current && typeof current === "object" && key in current) {
   819→      current = (current as Record<string, unknown>)[key]
   820→    } else {
   821→      return undefined
   822→    }
   823→  }
   824→  return typeof current === "string" ? current : undefined
   825→}
   826→
   827→export function translate(lang: Language, key: string, params?: Record<string, string | number>): string {
   828→  const dict = translations[lang] as unknown as Record<string, unknown>
   829→  const value = getNestedValue(dict, key)
   830→  if (!value) {
   831→    // Fall back to English
   832→    const enDict = translations.en as unknown as Record<string, unknown>
   833→    const enValue = getNestedValue(enDict, key)
   834→    if (!enValue) return key
   835→    return interpolate(enValue, params)
   836→  }
   837→  return interpolate(value, params)
   838→}
   839→
   840→function interpolate(s: string, params?: Record<string, string | number>): string {
   841→  if (!params) return s
   842→  return s.replace(/\{(\w+)\}/g, (_, k: string) => {
   843→    const v = params[k]
   844→    return v !== undefined ? String(v) : `{${k}}`
   845→  })
   846→}
