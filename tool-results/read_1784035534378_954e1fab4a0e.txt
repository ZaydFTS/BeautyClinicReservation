   883→// Helper to get a nested value from an object using a dot-path
   884→function getNestedValue(obj: unknown, path: string): string | undefined {
   885→  const keys = path.split(".")
   886→  let current: unknown = obj
   887→  for (const key of keys) {
   888→    if (current && typeof current === "object" && key in current) {
   889→      current = (current as Record<string, unknown>)[key]
   890→    } else {
   891→      return undefined
   892→    }
   893→  }
   894→  return typeof current === "string" ? current : undefined
   895→}
   896→
   897→export function translate(lang: Language, key: string, params?: Record<string, string | number>): string {
   898→  const dict = translations[lang] as unknown as Record<string, unknown>
   899→  const value = getNestedValue(dict, key)
   900→  if (!value) {
   901→    // Fall back to English
   902→    const enDict = translations.en as unknown as Record<string, unknown>
   903→    const enValue = getNestedValue(enDict, key)
   904→    if (!enValue) return key
   905→    return interpolate(enValue, params)
   906→  }
   907→  return interpolate(value, params)
   908→}
   909→
   910→function interpolate(s: string, params?: Record<string, string | number>): string {
   911→  if (!params) return s
   912→  return s.replace(/\{(\w+)\}/g, (_, k: string) => {
   913→    const v = params[k]
   914→    return v !== undefined ? String(v) : `{${k}}`
   915→  })
   916→}
